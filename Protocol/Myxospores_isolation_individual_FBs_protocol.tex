% Myxospores isolation
\subsection*{Protocol: Myxospores isolation}

\begin{itemize}[itemsep=-0.1em]
    \item Take pictures of FBs plates before and after harvest
    \item Harvest 2 times 20 FBs for each experimental conditions in:
    \begin{itemize}[itemsep=-0.2em, topsep=-0.2em]
        \item[-] 100 µL BSA (\textit{200 concentration condition}
        \item[-] 1 mL BSA (\textit{'Usual' condition)} 
    \end{itemize}
    \item  2 washing: centrifugation 10 min at $8000\times g$ with MM,
    \item Complete volume to 200 µL (\textit{200 concentration condition}) and 2 mL (\textit{Usual condition}),
    \item Sonication: \textit{Time} 2 min, \textit{Pulse} 10 sec ON / 15 sec OFF, \textit{Amplitude} 28\%, \textit{Probe} 60-62k,
    \item Heat Treatment 2 h,
    \item\underline{Only for Usual condition}: Concentration at 200 µL then homogenisation (2nd sonication) \textit{Time} 30 sec at same parameters (c.f. 1st sonication parameters above),
    \item Imaging at ×40 in phase contrast (and fluorescence) using Malassez slide (counting chamber)
\end{itemize}
